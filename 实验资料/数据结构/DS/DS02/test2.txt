#include <stdio.h>

// ʵ            Ԫ  ļӷ 
int add(int a[][3],int b[][3],int c[][3],int t1,int t2)
{
    int cnt=0;  //   Ӻ    Ԫ صĸ   
    int m=0,n=0;  // a  b  ָ  

    // ͬʱ            
    while(m<t1&&n<t2)
    {
        //  Ƚϵ ǰԪ ص λ  
        if(a[m][0]<b[n][0] || (a[m][0]==b[n][0] && a[m][1]<b[n][1]))
        {
            // a  Ԫ  λ ý С  ֱ Ӽ     
            c[cnt][0]=a[m][0];
            c[cnt][1]=a[m][1];
            c[cnt][2]=a[m][2];
            m++;
            cnt++;
        }
        else if(a[m][0]>b[n][0] || (a[m][0]==b[n][0] && a[m][1]>b[n][1]))
        {
            // b  Ԫ  λ ý С  ֱ Ӽ     
            c[cnt][0]=b[n][0];
            c[cnt][1]=b[n][1];
            c[cnt][2]=b[n][2];
            n++;
            cnt++;
        }
        else
        {
            // λ    ͬ    Ӻ      
            c[cnt][0]=a[m][0];
            c[cnt][1]=a[m][1];
            c[cnt][2]=a[m][2]+b[n][2];
            m++;
            n++;
            cnt++;
        }
    }

    //     a  ʣ  Ԫ  
    while(m<t1)
    {
        c[cnt][0]=a[m][0];
        c[cnt][1]=a[m][1];
        c[cnt][2]=a[m][2];
        m++;
        cnt++;
    }

    //     b  ʣ  Ԫ  
    while(n<t2)
    {
        c[cnt][0]=b[n][0];
        c[cnt][1]=b[n][1];
        c[cnt][2]=b[n][2];
        n++;
        cnt++;
    }

    return cnt;
}

int main()
{
    int n,m,t1,t2,num;
    scanf("%d %d %d %d", &n, &m, &t1, &t2);

    int a[t1][3];  //     a
    int b[t2][3];  //     b
    int max=t1>t2?t1:t2;
    int c[max*2][3];  //   Ӻ    

    //        aԪ  
    for(int i=0;i<t1;i++)
    {
        scanf("%d %d %d", &a[i][0], &a[i][1], &a[i][2]);
    }

    //        bԪ  
    for(int i=0;i<t2;i++)
    {
        scanf("%d %d %d", &b[i][0], &b[i][1], &b[i][2]);
    }

    // ִ о   ӷ 
    num=add(a,b,c,t1,t2);

    //           
    for(int i=0;i<num;i++)
    {
        printf("%d %d %d\n", c[i][0], c[i][1], c[i][2]);
    }

    return 0;
}
