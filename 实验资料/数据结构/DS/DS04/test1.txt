#include<stdio.h>
#define INF 10000

int main()
{
    int n;
    scanf("%d",&n);
    int g[101][101];
    //      ڽӾ   
    for(int i=0;i<n;i++)
        for(int j=0;j<n;j++)
            scanf("%d",&g[i][j]);
    int dist[101], path[101];
    //   ʼ      ͱ      
    for(int i=0;i<n;i++)
    {
        dist[i]=g[0][i]; //  ʼ    
        path[i]=0;       // 0  ʾδȷ     ·  
    }
    dist[0]=0;
    path[0]=1;
    // Dijkstra 㷨  ѭ  
    for(int cnt=1;cnt<n;cnt++)
    {
        int min=INF, u=-1;
        //   δȷ         
        for(int i=0;i<n;i++)
        {
            if(!path[i]&&dist[i]<min)
            {
                min=dist[i];
                u=i;
            }
        }
        if(u==-1)
            break;
        path[u]=1; //    Ϊ  ȷ  
        //           ľ   
        for(int v=0;v<n;v++)
        {
            if(!path[v]&&dist[v]>dist[u]+g[u][v])
            {
                dist[v]=dist[u]+g[u][v];
            }
        }
    }
    //    
    for(int i=0;i<n;i++)
        printf("%d\n", dist[i]);
    return 0;
}
