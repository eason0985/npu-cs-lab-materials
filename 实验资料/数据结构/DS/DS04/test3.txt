#include<stdio.h>
#define INF 10000

int main()
{
    int n;
    scanf("%d",&n);
    int g[101][101];
    //      ڽӾ   
    for(int i=0;i<n;i++)
        for(int j=0;j<n;j++)
            scanf("%d",&g[i][j]);
    // Floyd 㷨  ѭ           е     ·  
    for(int k=0;k<n;k++)
        for(int i=0;i<n;i++)
            for(int j=0;j<n;j++)
                if(g[i][k] + g[k][j] < g[i][j])
                    g[i][j] = g[i][k] + g[k][j];
    int m;
    scanf("%d",&m);
    int num[m];
    //     m   ѯ    ¼   ·      
    for(int i=0;i<m;i++)
    {
        int u, v;
        scanf("%d %d", &u, &v);
        num[i]=g[u][v];
    }
    //    ÿ   ѯ Ľ  
    for(int i=0;i<m;i++)
    {
        printf("%d\n",num[i]);
    }
    return 0;
}
